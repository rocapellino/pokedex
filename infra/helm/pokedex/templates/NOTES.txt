==============================================================================
  Pokédex Cloud Native Platform - Despliegue Exitoso con Helm
==============================================================================

1. Obtener la URL de la aplicación:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  - Frontend Web: http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}/
  - API Healthz:  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}/api/healthz
  - API Docs:     http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}/api/docs
{{- end }}
{{- else if contains "NodePort" .Values.web.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "pokedex.fullname" . }}-web-svc)
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.web.service.type }}
     NOTE: Puede tomar unos minutos hasta que la IP externa esté asignada.
           Puedes monitorear el estado con:
           kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "pokedex.fullname" . }}-web-svc
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "pokedex.fullname" . }}-web-svc --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.web.service.port }}
{{- else if contains "ClusterIP" .Values.web.service.type }}
  Para acceder localmente usando port-forwarding:
  - Frontend:
    kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "pokedex.fullname" . }}-web-svc 8080:{{ .Values.web.service.port }}
    Navegar a: http://localhost:8080/
  - API:
    kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "pokedex.fullname" . }}-api-svc 3000:{{ .Values.api.service.port }}
    Navegar a: http://localhost:3000/healthz
{{- end }}

2. Monitoreo y Verificación de Pods:
   kubectl get pods -n {{ .Release.Namespace }} -l app.kubernetes.io/instance={{ .Release.Name }}

3. Documentación y Operación:
   Consulta docs/runbooks/HELM_DEPLOYMENT_GUIDE.md para instrucciones avanzadas,
   escalamiento automático (HPA), perfiles de entorno y GitOps con ArgoCD.
